Docker image checks
===================

$ docker run --detach --name pain-counter-single -p 8081:8080 pain-counter:single
$ curl --silent http://localhost:8081/api/counter
{"count": 0}
$ curl --silent -X POST http://localhost:8081/api/counter
{"count": 1}
$ docker exec pain-counter-single id
uid=0(root) gid=0(root) groups=0(root)

$ docker run --detach --name pain-counter-multistage -p 8082:8080 pain-counter:multistage
$ curl --silent http://localhost:8082/api/counter
{"count": 0}
$ curl --silent -X POST http://localhost:8082/api/counter
{"count": 1}
$ docker exec pain-counter-multistage id
uid=0(root) gid=0(root) groups=0(root)

$ docker run --detach --name pain-counter-nonroot -p 8083:8080 pain-counter:nonroot
$ curl --silent http://localhost:8083/api/counter
{"count": 0}
$ curl --silent -X POST http://localhost:8083/api/counter
{"count": 1}
$ docker exec pain-counter-nonroot id
uid=1001(nagato) gid=1001(nagato) groups=1001(nagato),100(users)

$ docker image ls --filter reference='pain-counter:*'
IMAGE                       DISK USAGE   CONTENT SIZE
pain-counter:multistage     260MB        58.6MB
pain-counter:nonroot        261MB        58.6MB
pain-counter:single         390MB        91.6MB
